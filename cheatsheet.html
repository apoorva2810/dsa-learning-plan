Python cheatsheet for DSA

Everything on this page is standard library. These are the tools that come up again and again in the 250 problems.

Lists (arrays and stacks)

nums = [3, 1, 2]
nums.append(4)        # push, O(1)
nums.pop()            # pop last, O(1)      -> 4
nums.pop(0)           # pop FIRST, O(n)!    use deque instead
nums[-1]              # peek last
nums[1:3]             # slice copy [1, 2], O(k)
nums.sort()           # in place, O(n log n), stable
sorted(nums, key=lambda x: -x)        # new list, descending
sorted(pairs, key=lambda p: (p[0], -p[1]))   # sort by first asc, then second desc
nums.reverse(); nums[::-1]            # in place / copy
[0] * n               # n zeros
[[0] * cols for _ in range(rows)]     # 2-D grid (NOT [[0]*cols]*rows: shared rows!)
list(range(n))        # 0..n-1
for i, x in enumerate(nums): ...      # index + value
for a, b in zip(xs, ys): ...          # parallel iteration

Dictionaries and sets (hashing)

from collections import defaultdict, Counter

seen = set()
seen.add(x); x in seen; seen.remove(x); seen.discard(x)   # discard: no error if missing

d = {}
d.get(k, 0)                      # default if missing
d[k] = d.get(k, 0) + 1           # count
count = defaultdict(int)         # missing key -> 0
groups = defaultdict(list)       # missing key -> []
groups[key].append(item)

c = Counter("banana")            # {'a': 3, 'n': 2, 'b': 1}
c.most_common(2)                 # [('a', 3), ('n', 2)]
Counter(a) == Counter(b)         # anagram check

tuple(sorted(word))              # hashable key for a group of anagrams

Keys must be hashable: ints, strings, tuples — not lists. Convert list -> tuple when you need a list as a key.

Queues and deques

from collections import deque
q = deque([start])
q.append(x)           # push back
q.popleft()           # pop front, O(1)
q.appendleft(x); q.pop()         # it's double-ended
q[0]; q[-1]           # peek front / back

Heaps (priority queues)

import heapq
h = []                            # a plain list is the heap
heapq.heappush(h, (priority, item))
priority, item = heapq.heappop(h) # smallest priority first
h[0]                              # peek smallest
heapq.heapify(nums)               # O(n), in place
heapq.nlargest(k, nums); heapq.nsmallest(k, nums, key=...)

# max-heap: negate the keys
heapq.heappush(h, -x); largest = -heapq.heappop(h)

Only min-heaps exist. For a max-heap push -x. For ties, put a unique counter in the tuple so items are never compared.

Binary search

import bisect
i = bisect.bisect_left(a, x)   # first index with a[i] >= x
j = bisect.bisect_right(a, x)  # first index with a[j] > x
bisect.insort(a, x)            # insert keeping order, O(n)

# hand-rolled template: find the smallest value where condition(mid) is True
lo, hi = 0, n - 1
while lo < hi:
    mid = (lo + hi) // 2
    if condition(mid):
        hi = mid
    else:
        lo = mid + 1

Strings

s[::-1]                 # reverse
s.lower(); s.isalnum(); s.isdigit()
"".join(parts)          # build strings with a list + join, not += in a loop
ord("a"); chr(97)       # char <-> code
ord(c) - ord("a")       # 0..25 bucket index
s.split(); s.split(",") ; " ".join(words)

Strings are immutable: s += c in a loop is O(n²). Collect pieces in a list and "".join once.

Recursion and memoisation

from functools import lru_cache

@lru_cache(maxsize=None)        # or @cache in Python 3.9+
def f(i, j):
    if i == 0: return 1          # base case first
    return f(i - 1, j) + f(i, j - 1)

import sys
sys.setrecursionlimit(10_000)   # LeetCode's default is higher than CPython's 1000

Arguments must be hashable (no lists). Memoised recursion ("top-down DP") is usually the fastest route to a correct DP; convert to a table ("bottom-up") afterwards if asked.

Linked lists and trees (LeetCode classes)

class ListNode:
    def __init__(self, val=0, next=None):
        self.val, self.next = val, next

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

dummy = ListNode(0, head)       # dummy head: makes 'delete the first node' a normal case
slow = fast = head               # two pointers
while fast and fast.next:
    slow, fast = slow.next, fast.next.next   # slow stops at the middle

for dr, dc in DIRS:

adj = defaultdict(list)
for u, v in edges:
    adj[u].append(v); adj[v].append(u)    # undirected

visited = set()
def dfs(node):
    if node in visited: return
    visited.add(node)
    for nb in adj[node]: dfs(nb)

Bits

x & 1                 # last bit
x >> 1; x << 1        # halve / double
x & (x - 1)           # clears the lowest set bit (count bits, power-of-two test)
x ^ y                 # XOR: a ^ a == 0, a ^ 0 == a
bin(x).count("1")     # popcount (or x.bit_count() in 3.10+)
mask = 0xFFFFFFFF     # keep 32 bits when simulating fixed-width ints

Small things that save time

float("inf"), float("-inf")     # sentinels for min/max
a, b = b, a                      # swap
divmod(x, 10)                    # (x // 10, x % 10)
math.inf, math.gcd, math.isqrt
any(...), all(...), min(xs, key=...), max(xs, default=0)
itertools.accumulate(nums)       # prefix sums
